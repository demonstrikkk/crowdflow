# Optional Blueprint import engines (blueprint image enhancement).
#
# The import pipeline works without these (heuristic geometry + geometric
# labels only) but installs them for full fidelity:
#   - opencv-python-headless: line/edge detection for wall + gate extraction
#   - PyMuPDF: vector blueprints (PDF/SVG export from CAD tools)
#   - pytesseract: text labels (GATE A, EXIT 3, FOOD) -> semantic kinds
#
# pytesseract additionally requires the Tesseract OCR binary on the system:
#   https://github.com/tesseract-ocr/tesseract
#
# Install with:  pip install -r requirements-optional.txt

PyMuPDF==1.24.10
